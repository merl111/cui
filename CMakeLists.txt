cmake_minimum_required(VERSION 3.16)
project(cui VERSION 0.2.0 LANGUAGES C)

option(CUI_BUILD_EXAMPLES "Build the native settings example" ON)
option(CUI_BUILD_TESTS "Build layout and native integration tests" OFF)
option(CUI_BUILD_BINDING_TESTS "Build and test installed Python, Go, Zig and Rust examples (Linux)" OFF)
option(CUI_BUILD_SHARED "Build a shared library for FFI consumers" ON)

option(CUI_ENABLE_SANITIZERS "Instrument development builds with ASan/UBSan (Clang/GCC)" OFF)
option(CUI_BUILD_FUZZERS "Build Clang libFuzzer safety harnesses" OFF)
if(CUI_ENABLE_SANITIZERS)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "CUI_ENABLE_SANITIZERS requires Clang or GCC")
    endif()
    add_compile_options(-g -O1 -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
endif()

add_library(cui_objects OBJECT src/cui.c src/cui_layout.c src/cui_layouts.c src/cui_layout_algorithms.c src/cui_icons.c src/cui_draw.c src/cui_raster.c src/cui_controls.c src/cui_components.c src/cui_patterns.c src/cui_pattern_collections.c src/cui_pattern_filters.c src/cui_insights.c src/cui_desktop.c src/cui_files.c src/cui_keys.c src/cui_feedback.c src/cui_choices.c src/cui_search.c src/cui_tokens.c src/cui_navigation.c src/cui_breadcrumbs.c src/cui_inputs.c src/cui_datetime.c src/cui_tables.c)
set_target_properties(cui_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(cui_objects PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                      $<INSTALL_INTERFACE:include>)
target_compile_features(cui_objects PUBLIC c_std_99)
set_target_properties(cui_objects PROPERTIES C_EXTENSIONS OFF)
if(MSVC)
    # Modern MSVC enables its C99-compatible syntax in C11 mode.
    target_compile_features(cui_objects PUBLIC c_std_11)
    target_compile_options(cui_objects PRIVATE /W4)
else()
    target_compile_options(cui_objects PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    target_sources(cui_objects PRIVATE src/cui_win32.c src/cui_win32_icons.c src/cui_win32_desktop.c src/cui_win32_files.c src/cui_win32_tables.c src/cui_win32_pickers.c src/cui_win32_keys.c src/cui_win32_draw.c)
    target_compile_definitions(cui_objects PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(cui_objects PUBLIC comctl32 gdiplus user32 gdi32 dwmapi advapi32 ole32 oleacc shell32 uuid comdlg32)
elseif(APPLE)
    enable_language(OBJC)
    target_sources(cui_objects PRIVATE src/cui_macos_draw.m src/cui_macos.m src/cui_macos_desktop.m src/cui_macos_files.m src/cui_macos_pickers.m)
    set_source_files_properties(src/cui_macos_draw.m src/cui_macos.m src/cui_macos_desktop.m src/cui_macos_files.m src/cui_macos_pickers.m PROPERTIES COMPILE_OPTIONS "-fno-objc-arc")
    target_link_libraries(cui_objects PUBLIC "-framework AppKit" "-framework QuartzCore")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4-x11>=4.6 xext)
    target_sources(cui_objects PRIVATE src/cui_gtk.c src/cui_gtk_controls.c src/cui_gtk_desktop.c src/cui_gtk_files.c src/cui_gtk_layouts.c src/cui_gtk_navigation.c src/cui_gtk_tables.c src/cui_gtk_pickers.c src/cui_gtk_keys.c src/cui_gtk_draw.c)
    target_link_libraries(cui_objects PUBLIC PkgConfig::GTK4 m)
else()
    message(FATAL_ERROR "CUI supports Windows, macOS, and Linux")
endif()

add_library(cui STATIC)
target_link_libraries(cui PUBLIC cui_objects)
if(CUI_BUILD_SHARED)
    add_library(cui_shared SHARED)
    target_link_libraries(cui_shared PUBLIC cui_objects)
    set_target_properties(cui_shared PROPERTIES OUTPUT_NAME cui WINDOWS_EXPORT_ALL_SYMBOLS ON)
    if(WIN32)
        set_target_properties(cui_shared PROPERTIES ARCHIVE_OUTPUT_NAME cui_import)
    endif()
    install(TARGETS cui_shared LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
endif()

# xvfb-run's display-number allocation races when several wrappers start at once.
# Keep native processes isolated even when callers use ctest --parallel.
if(CUI_BUILD_TESTS AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    get_property(CUI_TESTS DIRECTORY PROPERTY TESTS)
    foreach(CUI_TEST IN LISTS CUI_TESTS)
        if(NOT CUI_TEST STREQUAL "layout")
            set_tests_properties(${CUI_TEST} PROPERTIES RESOURCE_LOCK cui_xvfb)
        endif()
    endforeach()
endif()

# Assertions contain native interactions in the C test programs. Keep them
# enabled even when a developer or CI selects a Release configuration.
if(CUI_BUILD_TESTS)
    get_property(CUI_BUILD_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(CUI_TEST_TARGET IN LISTS CUI_BUILD_TARGETS)
        if(CUI_TEST_TARGET MATCHES "_test$" OR CUI_TEST_TARGET STREQUAL "cui_showcase")
            if(MSVC)
                target_compile_options(${CUI_TEST_TARGET} PRIVATE /UNDEBUG)
            else()
                target_compile_options(${CUI_TEST_TARGET} PRIVATE -UNDEBUG)
            endif()
        endif()
    endforeach()
endif()
if(CUI_BUILD_FUZZERS)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang" OR NOT CUI_ENABLE_SANITIZERS)
        message(FATAL_ERROR "Fuzzing requires Clang and CUI_ENABLE_SANITIZERS=ON")
    endif()
    if(CUI_BUILD_SHARED OR CUI_BUILD_TESTS OR CUI_BUILD_EXAMPLES)
        message(FATAL_ERROR "Use a separate fuzz build with SHARED, TESTS and EXAMPLES disabled")
    endif()
    target_compile_options(cui_objects PRIVATE -fsanitize=fuzzer-no-link)
    foreach(FUZZ IN ITEMS icons models layout draw)
        add_executable(cui_fuzz_${FUZZ} tests/fuzz/${FUZZ}.c)
        target_include_directories(cui_fuzz_${FUZZ} PRIVATE src)
        target_link_libraries(cui_fuzz_${FUZZ} PRIVATE cui)
        target_compile_options(cui_fuzz_${FUZZ} PRIVATE -UNDEBUG -fsanitize=fuzzer-no-link)
        target_link_options(cui_fuzz_${FUZZ} PRIVATE -fsanitize=fuzzer)
    endforeach()
endif()
